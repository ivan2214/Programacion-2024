ALGORITMO ARREGLO_VOCALES_Y_FRECUENCIA

entradas: caracter: caracter (Cadena de caracteres con punto ".")
salidas: vocales: arreglo de caracteres, frecuencia: arreglo de enteros, totalVocales: entero

A0 inicializar
A1 PROCESAR_CADENA
A2 CONTAR_FRECUENCIA
A3 ESCRIBIR_RESULTADOS
A4 parar

A0
totalVocales <- 0
inicializar arreglo vocales[43] // Max 43 como en el ejemplo
inicializar arreglo frecuencia[5] <- {0, 0, 0, 0, 0} // a, e, i, o, u

A1 PROCESAR_CADENA
LEER(caracter)
MIENTRAS (caracter <> ".") HACER
SI (caracter = "a" ∨ caracter = "e" ∨ caracter = "i" ∨ caracter = "o" ∨ caracter = "u") ENTONCES
vocales[totalVocales] <- caracter
totalVocales <- totalVocales + 1
FIN_SI
LEER(caracter)
FIN_MIENTRAS

A2 CONTAR_FRECUENCIA
PARA cada vocal EN vocales HACER
SI (vocal = "a") ENTONCES
frecuencia[0] <- frecuencia[0] + 1
SINO SI (vocal = "e") ENTONCES
frecuencia[1] <- frecuencia[1] + 1
SINO SI (vocal = "i") ENTONCES
frecuencia[2] <- frecuencia[2] + 1
SINO SI (vocal = "o") ENTONCES
frecuencia[3] <- frecuencia[3] + 1
SINO SI (vocal = "u") ENTONCES
frecuencia[4] <- frecuencia[4] + 1
FIN_SI
FIN_PARA

A3 ESCRIBIR_RESULTADOS
escribir(vocales, totalVocales)
escribir(frecuencia[0], frecuencia[1], frecuencia[2], frecuencia[3], frecuencia[4])
